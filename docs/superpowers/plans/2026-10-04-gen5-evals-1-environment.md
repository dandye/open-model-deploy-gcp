# Gen 5 Evals, Plan 1 of 3: Data-Driven SocEnvironment Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make `SocEnvironment` fully data-driven (an `ips` inventory, an `events` list, host telemetry, and a `replace` mode) while proving, with a golden characterization test recorded before any change, that the frozen Gen 3 suite behaves byte-for-byte as it did.

**Architecture:** Every tool response becomes "inventory lookup first, then a fallback". The fallback has two implementations selected by the environment's `mode`: `overlay` (the Gen 3 default: load the built-in inventory, then keep the existing synthesized responses verbatim) and `replace` (Gen 5: start empty, respond only from data, and return `NO_DATA` / `NOT_FOUND` records for anything unknown). The two Gen 3 IP special cases move from code into a default `ips` inventory so they are data in both modes. A golden JSON captured from the current code is the oracle for every task in this plan.

**Tech Stack:** Python 3.13 (`.venv/bin/python`), dataclasses, `ipaddress`, pytest 8. No new dependencies.

**Spec:** `docs/superpowers/specs/2026-10-04-gen5-evals-design.md`, sections 3 and 7.1. Plan 2 (`2026-10-04-gen5-evals-2-suite.md`) builds the instancer, families, scoring, and reference policies on top of the interfaces defined here. Plan 3 (`2026-10-04-gen5-evals-3-runners-harbor.md`) wires runners, the hillclimber, and Harbor.

def run_probes(env: SocEnvironment, calls):
    """Executes probes in order and returns a JSON-serializable transcript."""
    transcript = []
    for kind, name, args in calls:
        if kind == "query":
            result = env.query_tool(name, args)
        elif kind == "action":
            result = env.execute_action(name, args)
        elif kind == "fault":
            env.inject_fault(name, **args)
            result = {"fault_injected": name}
        elif kind == "clear_faults":
            env.clear_faults()
            result = {"faults_cleared": True}
        else:
            raise AssertionError(f"unknown probe kind {kind}")
        transcript.append({"kind": kind, "name": name, "args": args, "result": result})
    transcript.append({
        "kind": "final_state",
        "blocked_indicators": sorted(env.blocked_indicators),
        "isolated_hosts": sorted(name for name, host in env.hosts.items() if host.isolated),
        "case_states": {
            case_id: [case.status, case.stage, len(case.comments)]
            for case_id, case in sorted(env.cases.items())
        },
    })
    return transcript


def canonical(obj) -> str:
    return json.dumps(obj, sort_keys=True, separators=(",", ":"), default=str)


def test_gen3_environment_matches_golden():
    env = SocEnvironment()
    actual = run_probes(env, probe_calls(env))

    if os.environ.get("GEN3_GOLDEN_RECORD") == "1":
        GOLDEN_PATH.parent.mkdir(parents=True, exist_ok=True)
        GOLDEN_PATH.write_text(json.dumps(actual, indent=2, sort_keys=True, default=str) + "\n", encoding="utf-8")
        pytest.skip(f"golden recorded to {GOLDEN_PATH}")

    assert GOLDEN_PATH.exists(), "golden file missing; run once with GEN3_GOLDEN_RECORD=1"
    expected = json.loads(GOLDEN_PATH.read_text(encoding="utf-8"))
    assert len(actual) == len(expected), "probe count changed; the probe list is part of the contract"
    for index, (got, want) in enumerate(zip(actual, expected)):
        assert canonical(got) == canonical(want), (
            f"probe {index} ({want.get('kind')} {want.get('name')} {want.get('args')}) diverged from golden:\n"
            f"  want: {canonical(want)[:400]}\n  got:  {canonical(got)[:400]}"
        )

@dataclass
class IpIndicatorEntity:
    """Represents an IP address indicator and its threat intelligence profile.

    Unknown IPs are never fabricated: an entity exists only when a world
    declares it. Optional descriptive fields are omitted from reports when
    empty so that overlay-mode reports keep their historical key sets.
    """

    ip: str
    reputation: str = "NO_DATA"
    verdict: str = "UNKNOWN"
    category: str = "Public IP"
    asn: str = "Unknown"
    cidr: str = ""
    routing_status: str = "Public Routable IP"
    threat_family: str = "None"
    malicious_votes: int = 0
    harmless_votes: int = 0
    vt_positives: int = 0
    vt_total: int = 72
    reverse_dns: str = ""
    service: str = ""
    host_association: str = ""
    country: str = ""
    extra: Dict[str, Any] = field(default_factory=dict)

    def to_report(self) -> Dict[str, Any]:
        """Renders the report dictionary returned by get_ip_address_report."""
        cidr = self.cidr or (f"{self.ip.rsplit('.', 1)[0]}.0/24" if self.ip.count(".") == 3 else "")
        report: Dict[str, Any] = {
            "ip": self.ip,
            "category": self.category,
            "reputation": self.reputation,
            "verdict": self.verdict,
            "malicious_votes": self.malicious_votes,
            "harmless_votes": self.harmless_votes,
            "asn": self.asn,
            "cidr": cidr,
            "routing_status": self.routing_status,
            "vt_positives": self.vt_positives,
            "vt_total": self.vt_total,
            "reverse_dns": self.reverse_dns,
        }
        if self.threat_family and self.threat_family != "None":
            report["threat_family"] = self.threat_family
        if self.service:
            report["service"] = self.service
        if self.host_association:
            report["host_association"] = self.host_association
        if self.country:
            report["country"] = self.country
        report.update(deepcopy(self.extra))
        return report
```

    @classmethod
    def from_eval_case(cls, case: Dict[str, Any]) -> "SocEnvironment":
        """Create an isolated case environment from its declared world and tool fault.

        Gen 5 cases carry a full ``environment_state`` (``mode: replace``) and are
        self-contained. Gen 3 cases carry none and get the overlay default.
        """
        state = case.get("environment_state")
        generation = int(case.get("generation") or 3)
        if state is None and generation >= 5:
            raise ValueError(
                f"case {case.get('eval_id', '<unknown>')} declares generation {generation} "
                "but has no environment_state"
            )
        if state is not None and not isinstance(state, dict):
            raise ValueError("environment_state must be a JSON object")
        environment = cls(initial_state=state)
        fault = case.get("fault_injection")
        if fault is not None:
            if not isinstance(fault, dict) or not isinstance(fault.get("tool"), str) or not fault["tool"].strip():
                raise ValueError("fault_injection must identify a nonempty tool name")
            environment.inject_fault(
                tool_name=fault["tool"],
                status_code=fault.get("status_code", 503),
                error=fault.get("error", "Service Unavailable"),
            )
        return environment
```
